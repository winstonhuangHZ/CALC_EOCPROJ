\begin{frame}{Introduction To DE Calculation Approaches}
    \begin{columns}[c]
        \column{0.55\textwidth}
        \begin{figure}
            \includegraphics[width=0.8\textwidth]{figures/Comparison_of_the_Runge-Kutta_methods_for_the_differential_equation_(red_is_the_exact_solution).svg.png}
            \caption{\footnotesize Comparison of the Runge-Kutta methods for the differential equation. Red line is the exact solution, 4th order Runge-Kutta is what we mainly use in this project, and is significantly more accurate. Figure from Wikimedia Commons, CC BY-SA 3.0 \ccbysa}
        \end{figure}
        
        \column{0.45\textwidth}
        \textbf{\small Fourth-Order Runge-Kutta (RK4) Approach: \\\vspace{1em}}
        \scalebox{0.8}{%
        \fbox{
        \vbox{
        \begin{align*}
            \text{Let}\ \frac{\mathrm{d}y}{\mathrm{d}x} &= f(x,y), \\
            y_{n+1} &= y_n + \frac{\Delta x}{6} (k_1 +2k_2+2k_3+k_4) \\
            \text{Where, } \\
            k_1 &= f(x_n, y_n) \\
            k_2 &= f(x_n+\frac{\Delta x}{2}, y_n+\frac{\Delta x}{2} k_1) \\
            k_3 &= f(x_n+\frac{\Delta x}{2}, y_n+\frac{\Delta x}{2} k_2) \\
            k_4 &= f(x_n+\Delta x, y_n+\Delta x k_3)
        \end{align*}
        }
        }
        }
    \end{columns} 
\end{frame}


\begin{frame}{Euler's Method: Simple DE Approximation}
    \begin{columns}[c]
        \column{0.55\textwidth}
        \begin{figure}
            \includegraphics[width=0.8\textwidth]{figures/Euler_method.svg.png}
            \caption{\footnotesize Illustration of the Euler method. The unknown curve is in blue, and its Euler approximation is in red. Figure from Wikimedia Commons, Public Domain. \ccLogo \ccZero}
        \end{figure}
        \column{0.4\textwidth}
        \begin{block}{\small Euler's Method:}
            \vspace{-1.25em}
            \begin{align*}
                \text{Let} \frac{\mathrm{d}y}{\mathrm{d}x} &= f(x,y), \\
                y_{n+1} &= y_n+\Delta x f(x_n,y_n)
            \end{align*}
            \small The key of Euler's Method is to use the slope at $f_n$ as the approximation slope of the entire step ($\Delta x$), it's quite like a common sense lol. So Euler's Method underestimates when the actual function concaves up and overestimates when the function concaves down.
        \end{block}

    \end{columns}
\end{frame}

\begin{frame}{Methodology and Experiment}
    \small The system of differential equations we use in this project can be written as:
    \\ \textit{(The derivation process is avaliable in the appendix).}

    %M = [ 2m_1+m_2-m_2\cos{(2\theta_1-2\theta_2)} ]
    \scalebox{0.78}{%
    \vbox{
    \begin{align*}
        \alpha_1 &= \frac{\mathrm{d}\omega_1}{\mathrm{d}t} = \frac{-g(2m_1+m_2)\sin{(\theta_1)}-m_2g\sin{(\theta_1-2\theta_2)}-2\sin{(\Delta)}m_2(\omega_2^2l_2+\omega^2_1l_1\cos{(\Delta)})}{l_1 \cdot \left[2m_1+m_2-m_2\cos{(2\theta_1-2\theta_2)}\right]}\\[10pt]
        \alpha_2 &= \frac{\mathrm{d}\omega_2}{\mathrm{d}t} = \frac{2\sin{(\Delta)}\left[ \omega_1^2 l_1 (m_1+m_2) +g(m_1+m_2)\cos{\theta_1} + w_2^2l_2m_2 \cos{\Delta} \right]}{l_2 \cdot \left[2m_1+m_2-m_2\cos{(2\theta_1-2\theta_2)}\right]} \\[10pt]
        \text{Where, } \\[5pt]
        \omega_1 &= \frac{\mathrm{d}\theta_1}{\mathrm{d}t},\,
        \omega_2 = \frac{\mathrm{d}\theta_2}{\mathrm{d}t},\,
        \Delta = \theta_1-\theta_2 
    \end{align*}
    }
    }

    \small In this system, $\theta_{\bullet}$, $\omega_{\bullet}$, and $\alpha_{\bullet}$ respectively represents angular position, angular speed and angular acceleration for each part of the pendulum. $m_{\bullet}$ is the symbol of the mass of each pendulum, and $l_{\bullet}$ is the length of each pendulum.
