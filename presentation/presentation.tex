\begin{frame}{Lagrangian Physics and Double Pendulum}

    \small Start from the Euler-Lagrangian Equation,

    \begin{equation}
    \frac{\partial L}{\partial q} - \frac{d}{dt} \frac{\partial L}{\partial \dot{q}} = 0
    \label{eq:euler-lagrangian}
    \end{equation}
    \\where the Lagrangian $L = T-V$, $\partial$ (in multivariable calculus) basically means what $\mathrm{d}$ represents in normal calculus, $q$ is the generalized coordinates, which is angle ($\theta_i$) in our problem.
    \\Citing~\ref{eq:euler-lagrangian}, for double pendulums, we have
    \begin{align}
        \frac{\mathrm{d}}{\mathrm{d}t} \left( \frac{\partial L}{\partial \dot{\theta}_1}\right) - \frac{\partial L}{\partial \theta_1} = 0,\,\frac{\mathrm{d}}{\mathrm{d}t} \left( \frac{\partial L}{\partial \dot{\theta}_2}\right) - \frac{\partial L}{\partial \theta_2} = 0
    \end{align}

\end{frame}


\begin{frame}{Lagrangian Physics and Double Pendulum}

    \small Now let's move on to physics, deriving expressions for energy.
    \begin{align*}
        \text{Position of Ball 1:}\ &x_1 = l_1 \sin{\theta_1},\, y_1 = -l_1 \cos{\theta_1} \\
        \text{Position of Ball 2:}\ &x_2 = x_1 + l_2 \sin{\theta_2},\, y_2 = y_1 -l_2 \cos{\theta_2}
    \end{align*}
    \\\small So,
    \begin{align*}
        \text{Speed of Ball 1}\ v_1 &= \frac{\dd x_1}{\dd t},\ \text{Speed of Ball 2}\ v_2 = \frac{\dd x_2}{\dd t}\\
        \text{Total } \text{E}_k &= \frac{1}{2} m_1 (v_1)^2+\frac{1}{2} m_2 (v_2)^2,\, \text{Total E}_p = m_1gy_1+m_2gy_2\\
        L&=T-V=\text{E}_k-\text{E}_p=\frac{1}{2}m_1(v_1)^2+\frac{1}{2}m_2(v_2)^2-m_1gy_1-m_2gy_2
    \end{align*}

\end{frame}

\begin{frame}{Double Pendulum Equation System Derivation}

    \small Substitute this Lagrangian $L$ expression back into formula (2) mentioned before, \\[5pt]
    For example, for $\frac{\partial L}{\partial \theta_1}$, the partial differentiation could be calculated as:
    \begin{align*}
        \frac{\partial L}{\partial \theta_1} = \frac{\partial}{\partial \theta_1} \bigg[ &\frac{1}{2} m_1 (l_1^2 \cos^2{\theta_1}\cdot\dot{\theta}_1^2+l_1^2\sin^2{\theta_1}\cdot\dot{\theta}_1^2)\\[3pt]
        +&\frac{1}{2} m_2 (A+B)\\
        -& m_1g\left( -l_1\,\cos{\theta_1} \right)-m_2g\left( -l_1\cos{\theta_1}-l_2\cos{\theta_2} \right) \bigg]\\
        A = l_1^2\cos^2&{\theta_1}\cdot\dot{\theta}_1^2+l_2^2\cos^2{\theta}_2\cdot\dot{\theta}_2^2+2\,l_1l_2\cos{\theta_1}\cos{\theta_2}\cdot\dot{\theta}_1\dot{\theta}_2,\\
        B = l_1^2\hspace{0.11em}\sin^2&{\theta_1}\cdot\dot{\theta}_1^2+l_2^2\hspace{0.11em}\sin^2{\theta}_2\cdot\dot{\theta}_2^2+2\,l_1l_2\hspace{0.11em}\sin{\theta_1}\hspace{0.11em}\sin{\theta_2}\cdot\dot{\theta}_1\dot{\theta}_2
    \end{align*}
    After repeating this procedure 4 times, the equation could be expanded and solved.

\end{frame}

        \column{0.45\textwidth}
        \textbf{\small Fourth-Order Runge-Kutta (RK4) Approach: \\\vspace{1em}}
        \scalebox{0.8}{%
        \fbox{
        \vbox{
        \begin{align*}
            \text{Let}\ \frac{\mathrm{d}y}{\mathrm{d}x} &= f(x,y), \\
            y_{n+1} &= y_n + \frac{\Delta x}{6} (k_1 +2k_2+2k_3+k_4) \\
            \text{Where, } \\
            k_1 &= f(x_n, y_n) \\
            k_2 &= f(x_n+\frac{\Delta x}{2}, y_n+\frac{\Delta x}{2} k_1) \\
            k_3 &= f(x_n+\frac{\Delta x}{2}, y_n+\frac{\Delta x}{2} k_2) \\
            k_4 &= f(x_n+\Delta x, y_n+\Delta x k_3)
        \end{align*}
        }
        }
        }
    \end{columns} 
\end{frame}


\begin{frame}{Euler's Method: Simple DE Approximation}
    \begin{columns}[c]
        \column{0.55\textwidth}
        \begin{figure}
            \includegraphics[width=0.8\textwidth]{figures/Euler_method.svg.png}
            \caption{\footnotesize Illustration of the Euler method. The unknown curve is in blue, and its Euler approximation is in red. Figure from Wikimedia Commons, Public Domain. \ccLogo \ccZero}
        \end{figure}
        \column{0.4\textwidth}
        \begin{block}{\small Euler's Method:}
            \vspace{-1.25em}
            \begin{align*}
                \text{Let} \frac{\mathrm{d}y}{\mathrm{d}x} &= f(x,y), \\
                y_{n+1} &= y_n+\Delta x f(x_n,y_n)
            \end{align*}
            \small The key of Euler's Method is to use the slope at $f_n$ as the approximation slope of the entire step ($\Delta x$), it's quite like a common sense lol. So Euler's Method underestimates when the actual function concaves up and overestimates when the function concaves down.
        \end{block}

    \end{columns}
\end{frame}

\begin{frame}{Methodology and Experiment}
    \small The system of differential equations we use in this project can be written as:
    \\ \textit{(The derivation process is avaliable in the appendix).}

    %M = [ 2m_1+m_2-m_2\cos{(2\theta_1-2\theta_2)} ]
    \scalebox{0.78}{%
    \vbox{
    \begin{align*}
        \alpha_1 &= \frac{\mathrm{d}\omega_1}{\mathrm{d}t} = \frac{-g(2m_1+m_2)\sin{(\theta_1)}-m_2g\sin{(\theta_1-2\theta_2)}-2\sin{(\Delta)}m_2(\omega_2^2l_2+\omega^2_1l_1\cos{(\Delta)})}{l_1 \cdot \left[2m_1+m_2-m_2\cos{(2\theta_1-2\theta_2)}\right]}\\[10pt]
        \alpha_2 &= \frac{\mathrm{d}\omega_2}{\mathrm{d}t} = \frac{2\sin{(\Delta)}\left[ \omega_1^2 l_1 (m_1+m_2) +g(m_1+m_2)\cos{\theta_1} + w_2^2l_2m_2 \cos{\Delta} \right]}{l_2 \cdot \left[2m_1+m_2-m_2\cos{(2\theta_1-2\theta_2)}\right]} \\[10pt]
        \text{Where, } \\[5pt]
        \omega_1 &= \frac{\mathrm{d}\theta_1}{\mathrm{d}t},\,
        \omega_2 = \frac{\mathrm{d}\theta_2}{\mathrm{d}t},\,
        \Delta = \theta_1-\theta_2 
    \end{align*}
    }
    }

    \small In this system, $\theta_{\bullet}$, $\omega_{\bullet}$, and $\alpha_{\bullet}$ respectively represents angular position, angular speed and angular acceleration for each part of the pendulum. $m_{\bullet}$ is the symbol of the mass of each pendulum, and $l_{\bullet}$ is the length of each pendulum.
\end{frame}
